services:
  db:
    image: postgres:16-alpine
    container_name: coinpilot_db
    restart: unless-stopped
    environment:
      POSTGRES_DB: coinpilot
      POSTGRES_USER: coinpilot
      POSTGRES_PASSWORD: coinpilot
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U coinpilot -d coinpilot"]
      interval: 5s
      timeout: 5s
      retries: 10

  web:
    build: .
    container_name: coinpilot_app
    restart: unless-stopped
    depends_on:
      db:
        condition: service_healthy
    env_file:
      - .env
    environment:
      PORT: "8000"
      DATABASE_URL: postgresql://coinpilot:coinpilot@db:5432/coinpilot
    ports:
      - "8000:8000"

  nginx:
    image: nginx:alpine
    container_name: coinpilot_proxy
    restart: unless-stopped
    ports:
      - "80:80"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - web
